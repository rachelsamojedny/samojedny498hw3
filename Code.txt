df = df.fillna('') # to avoid bug where there is missing values

#https://stackoverflow.com/questions/16476924/how-can-i-iterate-over-rows-in-a-pandas-dataframe
for _, row in df.iterrows(): # https://pandas.pydata.org/docs/reference/api/pandas.DataFrame.iterrows.html
  rowkey = str(row['DOL Vehicle ID']).encode() #since unique key we want this to encode the row
  row_in_table = DirectRow(rowkey, table) #https://cloud.google.com/python/docs/reference/bigtable/latest/google.cloud.bigtable.row.DirectRow
  #https://github.com/googleapis/google-cloud-python/issues/2517
  row_in_table.set_cell('ev_info', 'county', str(row['County']).encode('utf-8')) #need to encode after runnning into issues with special characters
  row_in_table.set_cell('ev_info', 'city', str(row['City']).encode('utf-8'))
  row_in_table.set_cell('ev_info', 'make', str(row['Make']).encode('utf-8'))
  row_in_table.set_cell('ev_info', 'model_year', str(row['Model Year']).encode('utf-8'))
  row_in_table.set_cell('ev_info', 'electric_range', str(row['Electric Range']).encode('utf-8'))
  row_in_table.commit() #inserts entire row with those attributes we wanted

from flask import Flask
from google.cloud import bigtable
from google.cloud.bigtable.row import ConditionalRow

btc = bigtable.Client(project= 'homework-3-455723', admin = True)
instance = btc.instance('ev-bigtable')
table = instance.table('ev-population')

app = Flask(__name__)

#used the big table api docs to find methods

@app.route('/') #test
def hello_world():
    return 'Hello World!'

@app.route('/rows')
def rows():
    rows = table.read_rows() #https://cloud.google.com/python/docs/reference/bigtable/latest/summary_method#google_cloud_bigtable_data_Table_read_rows_summary
    count = 0
    for _ in rows:
        count += 1
    return str(count)

@app.route('/Best-BMW')
def best_bmw():
    rows = table.read_rows()
    count = 0
    for row in rows:
        #print(row.cells)
        if (row.cell_value('ev_info', b'make') == b'BMW') and (float(row.cell_value('ev_info', b'electric_range').decode()) > 100): #big table stores everything in bytes
            count += 1
    return str(count)

@app.route('/tesla-owners')
def tesla_owners():
    rows = table.read_rows()
    count = 0
    for row in rows:
        if (row.cell_value('ev_info', b'make') == b'TESLA') and (row.cell_value('ev_info', b'city') == b'Seattle'):
          count += 1
    return str(count)

@app.route('/update')
def update():
    row = table.row(b'257246118')
    row.set_cell('ev_info', 'electric_range', '200') 
    row.commit()
    return "Success"

@app.route('/delete')
def delete():
    rows = table.read_rows()
    count = 0
    for row in rows:
        if (float(row.cell_value('ev_info', b'model_year').decode()) < 2014):
          direct_row = table.direct_row(row.row_key) #to be able to call delete
          direct_row.delete()  
          direct_row.commit()
    rows = table.read_rows() 
    count = 0
    for _ in rows:
        count += 1
    return str(count)

#https://cloud.google.com/run/docs/quickstarts/build-and-deploy/deploy-python-service
if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=80)
